// File: ChapterMiniApp.ts
// Bu mini uygulama öğrenci notu sorgulama isteğini uçtan uca modeller.
// İstemci, proxy, API ve backend ayrı sorumluluklar taşır.
// Çıktı, başarılı ve yetkisiz istek sonuçlarını gösterir.

type UygulamaIstegi = {
  yol: string;
  kullaniciAdi: string;
  yetkiliMi: boolean;
};

type UygulamaYaniti = {
  durumKodu: number;
  mesaj: string;
};

class NotBackendServisi {
  private notlar: Record = {
    "2026001": 88
  };

  notGetir(ogrenciNumarasi: string): UygulamaYaniti {
    const notDegeri = this.notlar[ogrenciNumarasi];

    if (notDegeri === undefined) {
      return { durumKodu: 404, mesaj: "Öğrenci kaydı bulunamadı." };
    }

    return {
      durumKodu: 200,
      mesaj: `${ogrenciNumarasi} numaralı öğrencinin notu: ${notDegeri}`
    };
  }
}

class NotApi {
  constructor(private backendServisi: NotBackendServisi) {}

  istekIsle(istek: UygulamaIstegi): UygulamaYaniti {
    const yolParcalari = istek.yol.split("/");
    const ogrenciNumarasi = yolParcalari[3];

    if (!ogrenciNumarasi) {
      return { durumKodu: 400, mesaj: "Öğrenci numarası eksik." };
    }

    return this.backendServisi.notGetir(ogrenciNumarasi);
  }
}

class TersVekil {
  constructor(private notApi: NotApi) {}

  yonlendir(istek: UygulamaIstegi): UygulamaYaniti {
    if (!istek.yetkiliMi) {
      return { durumKodu: 401, mesaj: "Kimlik doğrulaması gerekli." };
    }

    if (!istek.yol.startsWith("/api/notlar/")) {
      return { durumKodu: 404, mesaj: "Proxy için uygun yol bulunamadı." };
    }

    return this.notApi.istekIsle(istek);
  }
}

class TarayiciIstemcisi {
  constructor(private tersVekil: TersVekil) {}

  notSorgula(ogrenciNumarasi: string, yetkiliMi: boolean): void {
    const istek: UygulamaIstegi = {
      yol: `/api/notlar/${ogrenciNumarasi}`,
      kullaniciAdi: "ayse",
      yetkiliMi,
    };

    const yanit = this.tersVekil.yonlendir(istek);
    console.log(`[${yanit.durumKodu}] ${yanit.mesaj}`);
  }
}

const backendServisi = new NotBackendServisi();
const notApi = new NotApi(backendServisi);
const tersVekil = new TersVekil(notApi);
const istemci = new TarayiciIstemcisi(tersVekil);

istemci.notSorgula("2026001", true);
istemci.notSorgula("2026001", false);
istemci.notSorgula("2026999", true);

// Çıktı: [200] 2026001 numaralı öğrencinin notu: 88
// Çıktı: [401] Kimlik doğrulaması gerekli.
// Çıktı: [404] Öğrenci kaydı bulunamadı.
