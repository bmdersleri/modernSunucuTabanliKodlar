// Dosya: DosyaGuvenligiKontrolu.typescript
// izinVerilenTurler ve azamiBoyutBayt sabit bir politikayı temsil eder.
// guvenliDosyaAdiUret, orijinal ad yerine rastgele bir ad üreterek path traversal riskini ortadan kaldırır.
// Çıktı, geçerli ve geçersiz dosyalar için farklı sonuçların üretildiğini gösterir.

import { randomUUID } from "node:crypto";

interface YuklenenDosya {
  orijinalAd: string;
  mimeTuru: string;
  boyutBayt: number;
}

const izinVerilenTurler = new Set(["image/jpeg", "image/png"]);
const azamiBoyutBayt = 2 * 1024 * 1024; // 2 MB

interface DogrulamaSonucu {
  gecerliMi: boolean;
  hatalar: string[];
}

function dosyaGuvenlikDenetimiYap(dosya: YuklenenDosya): DogrulamaSonucu {
  const hatalar: string[] = [];

  if (!izinVerilenTurler.has(dosya.mimeTuru)) {
    hatalar.push(`Desteklenmeyen dosya türü: ${dosya.mimeTuru}`);
  }

  if (dosya.boyutBayt > azamiBoyutBayt) {
    hatalar.push(`Dosya boyutu sınırı aşıldı: ${dosya.boyutBayt} > ${azamiBoyutBayt} bayt`);
  }

  return { gecerliMi: hatalar.length === 0, hatalar };
}

function guvenliDosyaAdiUret(orijinalAd: string): string {
  const uzanti = orijinalAd.includes(".") ? orijinalAd.split(".").pop() : "bin";
  return `${randomUUID()}.${uzanti}`;
}

const gecerliDosya: YuklenenDosya = { orijinalAd: "kapak.jpg", mimeTuru: "image/jpeg", boyutBayt: 204_800 };
const zararliDosya: YuklenenDosya = { orijinalAd: "../../etc/passwd.png", mimeTuru: "image/png", boyutBayt: 3 * 1024 * 1024 };

console.log(dosyaGuvenlikDenetimiYap(gecerliDosya));
console.log(dosyaGuvenlikDenetimiYap(zararliDosya));
console.log(guvenliDosyaAdiUret(gecerliDosya.orijinalAd));
